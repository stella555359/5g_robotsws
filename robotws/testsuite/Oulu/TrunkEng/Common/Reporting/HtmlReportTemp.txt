        function preShow(elementId, show){
         var elem = document.getElementById(elementId);
         var elem2 = document.getElementById('div' + elementId);
         if(show && (elem.style.display == '' || elem.style.display == 'none')) {
          elem2.className='stepsPreview';
          elem2.style.display='block';
          try { // IE6 fix.
            elem.style.display='table-row';
          } catch(e) {
            elem.style.display='block';
          }
         }
         else if(elem2.className == 'stepsPreview') {
          elem2.className='steps';
          elem2.style.display='none';
          elem.style.display='none';
         }
        }
        function mouseOver(){
            document.getElementById("demo").style.color = "red";
        }
        function mouseOut(){
            document.getElementById("demo").style.color = "black";
        }
        function toggleDisplay(elementId) {
         preShow(elementId, false);
         var elem = document.getElementById(elementId);
         var elem2 = document.getElementById('div' + elementId);
         if(elem.style.display == 'block' || elem.style.display == 'table-row'){
          elem.style.display = 'none';
          elem2.style.display = 'none';
         } else {
          try {
            elem.style.display = 'table-row';
          } catch(e) {
            elem.style.display = 'block';
          }
          elem2.style.display = 'block';
         }
        }
        function toggleDisplay2(elemId1, elemId2) {
         var elem = document.getElementById(elemId1);
         var elem2 = document.getElementById(elemId2);
         if(elem2.innerHTML == '-'){
          elem.style.display = 'none';
          elem2.innerHTML = '+';
         } else {
          elem.style.display = 'block';
          elem2.innerHTML = '-';
         }
        }
        function collapseAll() {
        var elems = document.getElementsByClassName('trerror');
        for (var i = 0; i < elems.length; i++) {
        elems[i].style.display='none';
        }
        }
        function openAll() {
            var elems = document.getElementsByClassName('trerror');
            for (var i = 0; i < elems.length; i++) {
                elems[i].style.display = 'table-row';
                var divElem = document.getElementById('div' + elems[i].id);
                if (divElem) {
                    divElem.className = 'steps';
                    divElem.style.display = 'block';
                }
            }
        }
