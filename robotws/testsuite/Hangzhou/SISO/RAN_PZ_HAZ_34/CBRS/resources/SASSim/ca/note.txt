1. create root CA
openssl genrsa -out root-key.pem 2048
openssl req -new -x509 -days 3650 -key root-key.pem -out root-cert.pem

2. create server ca and issue by root ca
openssl genrsa -out server-key.pem 2048
openssl req -new -out server-req.csr -key server-key.pem
openssl x509 -req -in server-req.csr -out server-cert.pem -signkey server-key.pem -CA root-cert.pem -CAkey root-key.pem -CAcreateserial -days 3650

3. create expired server ca
openssl x509 -req -in server-req.csr -out server-expired-cert.pem -signkey server-key.pem -CA root-cert.pem -CAkey root-key.pem -CAcreateserial -days 1

4. check info
openssl x509 -noout -text -in root-cert.pem
openssl x509 -noout -text -in server-cert.pem
openssl x509 -noout -text -in server-expired-cert.pem

5. verify ca
5.1
openssl verify -verbose -CAfile root-cert.pem server-cert.pem
