Commands to start WTS:
cd /opt/wts/latest
sudo ./edaemon --port 22222
./egate --port 20000
	
After WTS/egate started, Setup NG/X2 from egate:
telnet 0 20000
cfg_sim file=H306_cell1_wts.cfg
enb_x2_link_up enb=1 target=2

//HR QCI-9 UE51 
ue_attach id=51
ue_dc_enb_start ue=51 target_gnb=2 target_cells=11 bearer=2 type=split_scg
trf_data_start ue=51 id=2 dir=ul mbps=10 trf_id=1

//add one QCI-6 DRB
bearer_add ue=51 id=3 filter=0 precedence=7 proto=17
enb_ue_dc_modify ue=51 bearer_to_add=3 by=enb cause=RN_action_desirable_for_radio_reasons
trf_data_start ue=51 id=3 dir=ul mbps=30 trf_id=1

//cqiDciCssAl1HR=10.6 < CQI=12 -> AL=1
csi_override_testmode_nr profile_id=1 cqi=12 cqi_mode=fixed cqi_min=1 cqi_max=15 ri_mode=fixed ri=0
ue_testmode_set_nr id=51 profile_id=1 action=activate testmode=csi_override scell=8
//cqiDciCssAl3HR=7.4 > CQI=7 > cqiDciCssAl4HR=4.5 -> AL=4
csi_override_testmode_nr profile_id=1 cqi=7 cqi_mode=fixed cqi_min=1 cqi_max=15 ri_mode=fixed ri=0

//cqiDciCssAl4HR=4.5 > CQI=4 > cqiDciCssAl8HR=2.2 -> AL=8
csi_override_testmode_nr profile_id=1 cqi=4 cqi_mode=fixed cqi_min=1 cqi_max=15 ri_mode=fixed ri=0





//release QCI-9
enb_ue_dc_modify ue=51 bearer_to_release=2 by=enb cause=RN_normal_release

//cqiDciCssAl1=10.1 > CQI=8.9 > cqiDciCssAl2=6.9 -> AL=2
csi_override_testmode_nr profile_id=1 cqi=7 cqi_mode=fixed cqi_min=1 cqi_max=15 ri_mode=fixed ri=0


//cqiDciCssAl3=6 > CQI=4 > cqiDciCssAl4=3.3 -> AL=4
csi_override_testmode_nr profile_id=1 cqi=4 cqi_mode=fixed cqi_min=1 cqi_max=15 ri_mode=fixed ri=0

ue_testmode_set_nr id=51 testmode=csi_override profile_id=1 action=deactivate



trf_data_stop ue=51

ue_detach id=51
