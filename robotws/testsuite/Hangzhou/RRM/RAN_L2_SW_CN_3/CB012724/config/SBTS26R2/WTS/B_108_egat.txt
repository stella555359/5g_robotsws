Commands to start WTS:
cd /opt/wts/latest
sudo ./edaemon --port 22222
./egate --port 20000
	
After WTS/egate started, Setup NG/X2 from egate:
telnet 0 20000
cfg_sim file=B108_cell11_wts.cfg
enb_x2_link_up enb=1 target=2  
//5UE CELL-11
ue_register id=1
set_pdu_param ue=1 pdu=1 trf_type=internal
ue_register id=2
set_pdu_param ue=2 pdu=1 trf_type=internal
ue_register id=3
set_pdu_param ue=3 pdu=1 trf_type=internal
ue_register id=4
set_pdu_param ue=4 pdu=1 trf_type=internal
ue_register id=5
set_pdu_param ue=5 pdu=1 trf_type=internal

ue_pdu_session_setup pdu=1 ue=1 nwslice=2
ue_pdu_session_setup pdu=1 ue=2 nwslice=3
ue_pdu_session_setup pdu=1 ue=3 nwslice=3
ue_pdu_session_setup pdu=1 ue=4 nwslice=2
ue_pdu_session_setup pdu=1 ue=5 nwslice=3
ue_pdu_session_modify ue=1 pdu=1 qosflows_to_add=2 by=amf

trf_data_start ue=1 pdu=1 qfi=1 dir=ul Kbps=20
trf_data_start ue=1 pdu=1 qfi=1 dir=dl Kbps=20
trf_data_start ue=1 pdu=1 qfi=2 dir=ul Kbps=20
trf_data_start ue=1 pdu=1 qfi=2 dir=dl Kbps=20
trf_data_start ue=2 pdu=1 qfi=1 dir=ul mbps=5
trf_data_start ue=2 pdu=1 qfi=1 dir=dl mbps=5
trf_data_start ue=3 pdu=1 qfi=1 dir=ul mbps=5
trf_data_start ue=3 pdu=1 qfi=1 dir=dl mbps=5
trf_data_start ue=4 pdu=1 qfi=1 dir=ul mbps=1
trf_data_start ue=4 pdu=1 qfi=1 dir=dl mbps=1
trf_data_start ue=5 pdu=1 qfi=1 dir=ul mbps=5
trf_data_start ue=5 pdu=1 qfi=1 dir=dl mbps=5

//step2 release UE2
trf_data_stop ue=2 pdu=1 qfi=1
ue_deregister id=2 ue_deregister_type=switch_off

//step3 UE2
ue_register id=2
set_pdu_param ue=2 pdu=1 trf_type=internal
ue_pdu_session_setup pdu=1 ue=2 nwslice=3
trf_data_start ue=2 pdu=1 qfi=1 dir=ul mbps=5
trf_data_start ue=2 pdu=1 qfi=1 dir=dl mbps=5

//step4 NACK/DTX
bler_testmode_nr profile_id=1 ul_bler=100,500,0,0,0,0,0,0 pdcch_misd_rate=10
ue_testmode_set_nr id=2 testmode=bler profile_id=1 action=activate
ue_testmode_set_nr id=5 testmode=bler profile_id=1 action=activate

trf_data_stop ue=1 pdu=1
trf_data_stop ue=2 pdu=1 qfi=1
trf_data_stop ue=3 pdu=1 qfi=1 
trf_data_stop ue=4 pdu=1 qfi=1
trf_data_stop ue=5 pdu=1 qfi=1
ue_testmode_set_nr id=2 testmode=bler profile_id=1 action=deactivate
ue_testmode_set_nr id=5 testmode=bler profile_id=1 action=deactivate

ue_deregister id=1 ue_deregister_type=switch_off
ue_deregister id=2 ue_deregister_type=switch_off
ue_deregister id=3 ue_deregister_type=switch_off
ue_deregister id=4 ue_deregister_type=switch_off
ue_deregister id=5 ue_deregister_type=switch_off
