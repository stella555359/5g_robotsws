Commands to start WTS:
cd /opt/wts/latest
sudo ./edaemon --port 22222
./egate --port 20000
	
After WTS/egate started, Setup NG/X2 from egate:
telnet 0 20000
cfg_sim file=H312_redcap_wts.cfg
enb_x2_link_up enb=1 target=2  

//UE2 REDCAP nonGBR 5QI8
ue_register id=1
set_pdu_param ue=1 pdu=1 trf_type=internal
ue_pdu_session_setup pdu=1 ue=1 nwslice=1 gw=1
ue_register id=2
set_pdu_param ue=2 pdu=1 trf_type=internal
ue_pdu_session_setup pdu=1 ue=2 nwslice=1 gw=1
ue_register id=3
set_pdu_param ue=3 pdu=1 trf_type=internal
ue_pdu_session_setup pdu=1 ue=3 nwslice=1 gw=1

//non-gbr traffic
trf_data_start ue=1 pdu=1 qfi=1 dir=ul trf_id=1 mbps=175
trf_data_start ue=1 pdu=1 qfi=1 dir=dl trf_id=1 mbps=175
trf_data_start ue=2 pdu=1 qfi=1 dir=ul trf_id=1 mbps=175
trf_data_start ue=2 pdu=1 qfi=1 dir=dl trf_id=1 mbps=175
trf_data_start ue=3 pdu=1 qfi=1 dir=ul trf_id=1 mbps=175
trf_data_start ue=3 pdu=1 qfi=1 dir=dl trf_id=1 mbps=175
//Stop after some seconds
trf_data_stop ue=1 pdu=1 qfi=1
trf_data_stop ue=2 pdu=1 qfi=1
trf_data_stop ue=3 pdu=1 qfi=1
ue_deregister id=1 ue_deregister_type=switch_off
ue_deregister id=2 ue_deregister_type=switch_off
ue_deregister id=3 ue_deregister_type=switch_off
